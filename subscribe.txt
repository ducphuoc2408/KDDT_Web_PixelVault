function subscribeToRentalRealtime(rentalId) {
  if (!window.supabaseClient) return;

  const channel = window.supabaseClient
    .channel('schema-db-changes')
    .on(
      'postgres_changes',
      {
        event: 'UPDATE',
        schema: 'public',
        table: 'rentals',
        filter: `id=eq.${rentalId}`
      },
      (payload) => {
        const newRecord = payload.new;
        if (newRecord.status === 'active' || newRecord.status === 'completed') {
          const statusText = document.getElementById('qr-status-text');
          if (statusText) {
            statusText.style.color = 'var(--success)';
            statusText.innerHTML = 'Thanh toán thành công! <svg style="width:16px;height:16px;vertical-align:middle;" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><polyline points="20 6 9 17 4 12"/></svg>';
            showToast('Thanh toán thành công. Đơn thuê đã được ghi nhận!', 'success');
            setTimeout(() => {
              closeModal('qr-modal');
            }, 3000);
          }
        }
      }
    )
    .subscribe();
}
</script>
