function processRentalDeposit() {
  if (!Store.currentUser) {
    showToast('Vui lòng đăng nhập để đặt thuê', 'info');
    openModal('auth-modal');
    return;
  }

  const rInfo = currentProduct.rentalInfo;
  if (!rInfo) return;

  const btnProcess = document.getElementById('btn-process-rental');
  btnProcess.innerHTML = '<svg style="width:20px;height:20px;animation:spin 1s linear infinite;" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><circle cx="12" cy="12" r="10" stroke-dasharray="32" stroke-dashoffset="0"/></svg> Đang xử lý...';
  btnProcess.disabled = true;

  try {
    const rentalData = {
      user_id: Store.currentUser.id,
      product_id: currentProduct.id,
      start_date: rInfo.startDate,
      end_date: rInfo.endDate,
      rental_days: rInfo.days,
      daily_rate: rInfo.dailyRate,
      deposit_amount: rInfo.depositAmount,
      total_price: rInfo.totalCheckout,
      status: 'pending_deposit'
    };

    const res = await window.SupabaseDB.createRental(rentalData);
    
    if (res && res.id) {
      const shortId = res.id.split('-')[0];
      const amount = rInfo.totalCheckout;
      const qrUrl = `https://img.vietqr.io/image/BIDV-96247HH125-compact2.png?amount=${Math.round(amount)}&addInfo=Thanh%20toan%20${shortId}&accountName=NGUYEN%20DUC%20PHUOC`;

      showQRModal(qrUrl, amount, addInfo, res.id);
    } else {
      showToast('Không thể tạo đơn thuê. Vui lòng thử lại.', 'error');
    }
  } catch (error) {
    console.error('Error processRentalDeposit:', error);
    showToast('Lỗi hệ thống', 'error');
  }

  btnProcess.innerHTML = 'Thanh toán ngay';
  btnProcess.disabled = false;
}
