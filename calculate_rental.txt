function calculateRental() {
  const startDateVal = document.getElementById('rent-start-date').value;
  const endDateVal = document.getElementById('rent-end-date').value;
  const summaryDiv = document.getElementById('rental-summary');
  const btnProcess = document.getElementById('btn-process-rental');

  if (!startDateVal || !endDateVal) {
    // summaryDiv.style.display = 'none'; // Keep visible
    btnProcess.disabled = true;
    return;
  }

  const start = new Date(startDateVal);
  const end = new Date(endDateVal);
  const timeDiff = end.getTime() - start.getTime();
  const days = Math.ceil(timeDiff / (1000 * 3600 * 24)) + 1;

  if (days <= 0) {
    showToast('Ngày kết thúc phải lớn hơn hoặc bằng ngày bắt đầu', 'error');
    // summaryDiv.style.display = 'none'; // Keep visible
    btnProcess.disabled = true;
    return;
  }

  const dailyRate = Math.round(productPrice * 0.05);
  const totalRental = dailyRate * days;
